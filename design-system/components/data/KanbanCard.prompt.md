```jsx
<KanbanCard customer="Linh Dan" phone="09123••••" item="Bánh Kem Dâu 22cm" channel="Zalo"
  badges={[<Badge tone="warning">SLA 08:43</Badge>, <Badge tone="primary">VIP</Badge>]} />
```
